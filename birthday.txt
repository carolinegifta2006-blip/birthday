// Smoothly begin the birthday experience.
function beginSurprise() {
  document.getElementById("message").scrollIntoView({
    behavior: "smooth"
  });
  celebrate(45);
}

// Photo gallery: display up to 3 selected photos.
const photoInput = document.getElementById("photos");
let photoURLs = [];

photoInput.addEventListener("change", function() {
  photoURLs.forEach(url => URL.revokeObjectURL(url));
  photoURLs = [];

  const files = Array.from(this.files || [])
    .filter(file => file.type.startsWith("image/"))
    .slice(0, 3);

  const cards = document.querySelectorAll(".photo");

  cards.forEach((card, index) => {
    const img = card.querySelector("img");

    if (files[index]) {
      const url = URL.createObjectURL(files[index]);
      photoURLs.push(url);
      img.src = url;
      card.classList.add("has-image");
    } else {
      img.removeAttribute("src");
      card.classList.remove("has-image");
    }
  });

  if (this.files.length > 3) {
    alert("Only the first 3 photos will be displayed.");
  }
});

// Birthday wish interaction.
let wished = false;

function makeWish() {
  const message = document.getElementById("cakeMessage");

  if (!wished) {
    wished = true;
    message.textContent =
      "Wish sent to the universe! ✨ May beautiful things find their way to you, Swetha. Happy Birthday! 💛";

    document.getElementById("cake").textContent = "🎂💖";
    celebrate(180);
  } else {
    message.textContent =
      "Keep that wish close, Swetha. Here's to a year full of lovely surprises! ✨";

    celebrate(70);
  }
}

// Confetti animation.
function celebrate(count = 160) {
  const colors = [
    "#f7cf83",
    "#f5a8bd",
    "#fff4df",
    "#bba9f4",
    "#a8e6d7",
    "#e9a9ff"
  ];

  for (let i = 0; i < count; i++) {
    const piece = document.createElement("div");

    piece.style.position = "fixed";
    piece.style.zIndex = "9999";
    piece.style.top = "-15px";
    piece.style.left = Math.random() * 100 + "vw";
    piece.style.width = 5 + Math.random() * 8 + "px";
    piece.style.height = 8 + Math.random() * 12 + "px";
    piece.style.background =
      colors[Math.floor(Math.random() * colors.length)];
    piece.style.pointerEvents = "none";

    document.body.appendChild(piece);

    const drift = (Math.random() - 0.5) * 260;
    const duration = 2000 + Math.random() * 2000;
    const rotation = Math.random() * 800;

    const animation = piece.animate([
      {
        transform: "translate(0, 0) rotate(0deg)",
        opacity: 1
      },
      {
        transform:
          `translate(${drift}px, ${innerHeight + 60}px) rotate(${rotation}deg)`,
        opacity: 0
      }
    ], {
      duration: duration,
      easing: "ease-out"
    });

    animation.onfinish = () => piece.remove();
  }
}

// Animated starry background.
const canvas = document.getElementById("stars");
const ctx = canvas.getContext("2d");

let width, height, stars = [];

function resizeStars() {
  const dpr = Math.min(window.devicePixelRatio || 1, 2);

  width = window.innerWidth;
  height = window.innerHeight;

  canvas.width = width * dpr;
  canvas.height = height * dpr;
  canvas.style.width = width + "px";
  canvas.style.height = height + "px";

  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

  stars = Array.from(
    { length: Math.min(140, Math.floor(width * height / 7000)) },
    () => ({
      x: Math.random() * width,
      y: Math.random() * height,
      r: 0.5 + Math.random() * 1.3,
      phase: Math.random() * 6,
      speed: 0.01 + Math.random() * 0.02
    })
  );
}

function drawStars() {
  ctx.clearRect(0, 0, width, height);

  stars.forEach(star => {
    star.phase += star.speed;

    const alpha = 0.3 + 0.7 *
      ((Math.sin(star.phase) + 1) / 2);

    ctx.beginPath();
    ctx.arc(star.x, star.y, star.r, 0, Math.PI * 2);
    ctx.fillStyle = `rgba(255,239,220,${alpha})`;
    ctx.fill();
  });

  requestAnimationFrame(drawStars);
}

window.addEventListener("resize", resizeStars);
resizeStars();
drawStars();
